import bundlebase.sync as bb

bundle = (bb.create("s3://data/sales-q1")
    .attach("jan.csv")
    .attach("feb.csv")
    .drop_column("ssn")
    .filter("status = 'closed_won'"))
bundle.commit("Q1 export, PII removed")
